pi_hyperforge_matterweaver = {
	category = specialist

	swappable_data = {
		default = {
			condition_string = SPECIALIST_JOB_TRIGGER
			building_icon = pi_deep_mantle_exotic_hyperforge
			icon = pi_hyperforge_matterweaver
		}
	}

	tags = { strategic_resource }

	possible_precalc = can_fill_specialist_job

	possible_pre_triggers = {
		has_owner = yes
		has_planet = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible = {
		owner = { is_regular_empire = yes }
		is_sapient = yes
	}

	resources = {
		category = planet_jobs
		produces = {
			sr_dark_matter = 0.75
            sr_living_metal = 0.75
		}
		upkeep = {
            alloys = 2
			energy = 3
            minerals = 4
		}
	}

	inline_script = "jobs/job_hyperforge_matterweaver_modifiers"

	weight = {
		weight = @specialist_job_weight
		mult = value:job_weights_modifier|JOB|pi_hyperforge_matterweaver|RESOURCE|sr_dark_matter|

		modifier = {
			factor = 1.5
			has_trait = trait_intelligent
		}
	}
}
